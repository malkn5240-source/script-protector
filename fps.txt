local function __isExecutor()
    local c = 0
    if type(getgenv) == "function" then c = c + 1 end
    if type(identifyexecutor) == "function" then c = c + 1 end
    if type(hookfunction) == "function" then c = c + 1 end
    if type(getrawmetatable) == "function" then c = c + 1 end
    if type(setclipboard) == "function" then c = c + 1 end
    if type(request) == "function" or type(http_request) == "function" then c = c + 1 end
    return c >= 4
end
local function __isRoblox()
    return (type(game) == "userdata" or type(game) == "table")
        and type(game.GetService) == "function"
        and type(workspace) ~= "nil"
        and type(Players) ~= "nil"
end
if not __isExecutor() or not __isRoblox() then
    return "no"
end

local function isTeammate(plr)
    if not State.teamCheck then return false end
    if not LP.Team or not plr.Team then return false end
    return LP.Team == plr.Team
end

local function getTarget()
    local char = LP.Character
    if not char then return nil end
    local myRoot = char:FindFirstChild("HumanoidRootPart")
    if not myRoot then return nil end
    local closest, minDist = nil, State.fov
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= LP and plr.Character and not isTeammate(plr) then
            local tRoot = plr.Character:FindFirstChild("HumanoidRootPart")
            local tHum = plr.Character:FindFirstChildOfClass("Humanoid")
            if tRoot and tHum and tHum.Health > 0 then
                local screenPos, onScreen = Camera:WorldToViewportPoint(tRoot.Position)
                if onScreen then
                    local cx, cy = Camera.ViewportSize.X / 2, Camera.ViewportSize.Y / 2
                    local dist = math.sqrt((screenPos.X - cx)^2 + (screenPos.Y - cy)^2)
                    if dist < minDist then
                        minDist = dist
                        closest = plr.Character
                    end
                end
            end
        end
    end
    return closest
end

local function removeESP(plr)
    if State.highlights[plr] then
        pcall(function() State.highlights[plr]:Destroy() end)
        State.highlights[plr] = nil
    end
    if State.billboards[plr] then
        pcall(function() State.billboards[plr]:Destroy() end)
        State.billboards[plr] = nil
    end
    if State.tracers[plr] then
        pcall(function() State.tracers[plr].Visible = false; State.tracers[plr]:Remove() end)
        State.tracers[plr] = nil
    end
    if State.boxes[plr] then
        for _, line in ipairs(State.boxes[plr]) do
            pcall(function() line.Visible = false; line:Remove() end)
        end
        State.boxes[plr] = nil
    end
end

local function clearAllESP()
    for plr, _ in pairs(State.highlights) do removeESP(plr) end
    State.highlights = {}
    State.billboards = {}
    State.tracers = {}
    State.boxes = {}
end

local function makeTracer(plr)
    if not Drawing or State.tracers[plr] then return end
    local line = Drawing.new("Line")
    line.Color = Color3.fromRGB(255, 255, 255)
    line.Thickness = 1.5
    line.Transparency = 0.8
    line.Visible = false
    State.tracers[plr] = line
end

local function makeBox(plr)
    if not Drawing or State.boxes[plr] then return end
    local lines = {}
    for i = 1, 4 do
        local line = Drawing.new("Line")
        line.Color = Color3.fromRGB(255, 255, 255)
        line.Thickness = 2
        line.Transparency = 0.9
        line.Visible = false
        table.insert(lines, line)
    end
    State.boxes[plr] = lines
end

local function updateBox(plr, char)
    if not Drawing or not State.showBox then return end
    if not State.boxes[plr] then makeBox(plr) end
    local lines = State.boxes[plr]
    if not lines then return end
    local root = char:FindFirstChild("HumanoidRootPart")
    if not root then return end
    local pos, onScreen = Camera:WorldToViewportPoint(root.Position)
    if not onScreen or pos.Z <= 0 then
        for _, line in ipairs(lines) do line.Visible = false end
        return
    end
    local scale = Camera.ViewportSize.Y / (2 * math.tan(math.rad(Camera.FieldOfView / 2))) / pos.Z
    local height = 3.5 * scale
    local width = 2 * scale
    local x = pos.X - width / 2
    local y = pos.Y - height / 2
    lines[1].From = Vector2.new(x, y); lines[1].To = Vector2.new(x + width, y); lines[1].Visible = true
    lines[2].From = Vector2.new(x + width, y); lines[2].To = Vector2.new(x + width, y + height); lines[2].Visible = true
    lines[3].From = Vector2.new(x + width, y + height); lines[3].To = Vector2.new(x, y + height); lines[3].Visible = true
    lines[4].From = Vector2.new(x, y + height); lines[4].To = Vector2.new(x, y); lines[4].Visible = true
end

-- ESP Loop
RunService.RenderStepped:Connect(function()
    if not State.esp then return end
    local myChar = LP.Character
    local myRoot = myChar and myChar:FindFirstChild("HumanoidRootPart")
    if not myRoot then return end
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= LP and plr.Character then
            local char = plr.Character
            local head = char:FindFirstChild("Head")
            local hum = char:FindFirstChildOfClass("Humanoid")
            local tRoot = char:FindFirstChild("HumanoidRootPart")
            if head and hum and hum.Health > 0 and tRoot then
                local dist = (myRoot.Position - tRoot.Position).Magnitude
                if dist > State.maxDistance then
                    if State.highlights[plr] then State.highlights[plr].Enabled = false end
                    if State.billboards[plr] then State.billboards[plr].Enabled = false end
                    if State.tracers[plr] then State.tracers[plr].Visible = false end
                    if State.boxes[plr] then for _, line in ipairs(State.boxes[plr]) do line.Visible = false end end
                else
                    if not State.highlights[plr] or not State.highlights[plr].Parent then addESP(plr) end
                    if State.highlights[plr] then State.highlights[plr].Enabled = not isTeammate(plr) end
                    local bb = State.billboards[plr]
                    if bb then
                        bb.Enabled = true
                        local nameLbl = bb:FindFirstChild("Name")
                        local distLbl = bb:FindFirstChild("Dist")
                        local hpLbl = bb:FindFirstChild("HP")
                        if nameLbl then nameLbl.Visible = State.showName; nameLbl.Text = plr.Name end
                        if distLbl then distLbl.Visible = State.showDistance; distLbl.Text = math.floor(dist) .. "m" end
                        if hpLbl then
                            hpLbl.Visible = State.showHealth
                            hpLbl.Text = "HP: " .. math.floor(hum.Health)
                            local r = hum.Health / hum.MaxHealth
                            hpLbl.TextColor3 = Color3.fromRGB(255 * (1-r), 255 * r, 100)
                        end
                    end
                    if State.showTracer and Drawing then
                        if not State.tracers[plr] then makeTracer(plr) end
                        local tr = State.tracers[plr]
                        if tr then
                            local sp, os = Camera:WorldToViewportPoint(tRoot.Position)
                            if os and sp.Z > 0 then
                                tr.From = Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y)
                                tr.To = Vector2.new(sp.X, sp.Y)
                                tr.Visible = true
                            else
                                tr.Visible = false
                            end
                        end
                    end
                    if State.showBox then updateBox(plr, char) end
                end
            else
                if State.highlights[plr] then State.highlights[plr].Enabled = false end
                if State.billboards[plr] then State.billboards[plr].Enabled = false end
                if State.tracers[plr] then State.tracers[plr].Visible = false end
                if State.boxes[plr] then for _, line in ipairs(State.boxes[plr]) do line.Visible = false end end
            end
        end
    end
end)

-- Auto Refresh
local function refreshPlayer(plr)
    if plr == LP or not plr.Character then return end
    removeESP(plr)
    task.wait(0.1)
    addESP(plr)
end

Players.PlayerAdded:Connect(function(plr)
    plr.CharacterAdded:Connect(function()
        task.wait(0.5)
        if State.esp then refreshPlayer(plr) end
    end)
end)
Players.PlayerRemoving:Connect(removeESP)

task.spawn(function()
    while task.wait(1.5) do
        if not State.esp then continue end
        for _, plr in ipairs(Players:GetPlayers()) do
            if plr ~= LP then
                if plr.Character then
                    local hum = plr.Character:FindFirstChildOfClass("Humanoid")
                    if hum and hum.Health > 0 then
                        if not State.highlights[plr] or not State.highlights[plr].Parent then refreshPlayer(plr) end
                    else
                        if State.highlights[plr] then removeESP(plr) end
                    end
                else
                    if State.highlights[plr] then removeESP(plr) end
                end
            end
        end
    end
end)

-- Aim Lock + Body Lock + Silent Aim
RunService.RenderStepped:Connect(function()
    if not State.aimLock and not State.bodyLock and not State.silentAim then return end
    local char = LP.Character
    if not char then return end
    local head = char:FindFirstChild("Head")
    local myRoot = char:FindFirstChild("HumanoidRootPart")
    local hum = char:FindFirstChildOfClass("Humanoid")
    if not head or not myRoot or not hum then return end
    if hum.Health <= 0 then return end
    
    local target = getTarget()
    if not target then return end
    local tRoot = target:FindFirstChild("HumanoidRootPart")
    if not tRoot then return end
    
    local dir = tRoot.Position - myRoot.Position
    local flatDir = Vector3.new(dir.X, 0, dir.Z)
    if flatDir.Magnitude < 0.1 then return end
    
    if State.bodyLock then
        pcall(function()
            head.CFrame = head.CFrame:Lerp(CFrame.new(head.Position, head.Position + flatDir.Unit), State.smooth)
        end)
    end
    
    if hum.AutoRotate ~= false then hum.AutoRotate = false end
    local rootPos = myRoot.Position
    local newCF = CFrame.lookAt(rootPos, rootPos + flatDir.Unit)
    myRoot.CFrame = myRoot.CFrame:Lerp(newCF, State.smooth)
    
    if State.aimLock or State.silentAim then
        if Camera.CameraSubject == head or Camera.CameraSubject == hum then
            local camPos = Camera.CFrame.Position
            Camera.CFrame = Camera.CFrame:Lerp(CFrame.lookAt(camPos, camPos + flatDir.Unit), State.smooth)
        end
    end
end)

-- Auto Shoot + Trigger Bot
RunService.Heartbeat:Connect(function()
    if not State.autoShoot and not State.triggerBot then return end
    local target = getTarget()
    if not target then return end
    local char = LP.Character
    if not char then return end
    
    local tool = char:FindFirstChildOfClass("Tool")
    if tool then
        pcall(function() tool:Activate() end)
    end
    
    if State.triggerBot then
        local tRoot = target:FindFirstChild("HumanoidRootPart")
        if tRoot then
            local sp = Camera:WorldToViewportPoint(tRoot.Position)
            local cx, cy = Camera.ViewportSize.X / 2, Camera.ViewportSize.Y / 2
            local dist = math.sqrt((sp.X - cx)^2 + (sp.Y - cy)^2)
            if dist < 10 then
                pcall(function()
                    if mouse1click then mouse1click() end
                end)
            end
        end
    end
end)

-- Walk Speed + Jump Power
RunService.Heartbeat:Connect(function()
    local char = LP.Character
    if not char then return end
    local hum = char:FindFirstChildOfClass("Humanoid")
    if not hum then return end
    if State.walkSpeed then
        hum.WalkSpeed = State.walkSpeedValue
    else
        hum.WalkSpeed = State.origWalkSpeed
    end
    if State.jumpPower then
        hum.JumpPower = State.jumpPowerValue
    elseif hum.UseJumpPower then
        hum.JumpPower = State.origJumpPower
    end
end)

-- Infinite Jump
UIS.JumpRequest:Connect(function()
    if not State.infiniteJump then return end
    local char = LP.Character
    if not char then return end
    local hum = char:FindFirstChildOfClass("Humanoid")
    if hum then hum:ChangeState(Enum.HumanoidStateType.Jumping) end
end)

RunService.Heartbeat:Connect(function()
    if not State.infiniteJump then return end
    local char = LP.Character
    if not char then return end
    local hum = char:FindFirstChildOfClass("Humanoid")
    local root = char:FindFirstChild("HumanoidRootPart")
    if not hum or not root then return end
    if hum:GetState() == Enum.HumanoidStateType.Freefall or hum:GetState() == Enum.HumanoidStateType.Jumping then
        if UIS:IsKeyDown(Enum.KeyCode.Space) then
            root.Velocity = Vector3.new(root.Velocity.X, 55, root.Velocity.Z)
        end
    end
end)

-- Fly
UIS.InputBegan:Connect(function(input)
    if input.KeyCode == Enum.KeyCode.F then
        if State.fly then
            State.fly = false
            if State.flyVelocity then State.flyVelocity:Destroy(); State.flyVelocity = nil end
        else
            State.fly = true
            local char = LP.Character
            if not char then return end
            local root = char:FindFirstChild("HumanoidRootPart")
            if not root then return end
            State.flyVelocity = Instance.new("BodyVelocity")
            State.flyVelocity.MaxForce = Vector3.new(9e9, 9e9, 9e9)
            State.flyVelocity.Velocity = Vector3.zero
            State.flyVelocity.Parent = root
        end
    end
end)

RunService.Heartbeat:Connect(function()
    if not State.fly then return end
    local char = LP.Character
    if not char then return end
    local root = char:FindFirstChild("HumanoidRootPart")
    if not root or not State.flyVelocity then return end
    local dir = Vector3.zero
    if UIS:IsKeyDown(Enum.KeyCode.W) then dir = dir + Camera.CFrame.LookVector end
    if UIS:IsKeyDown(Enum.KeyCode.S) then dir = dir - Camera.CFrame.LookVector end
    if UIS:IsKeyDown(Enum.KeyCode.A) then dir = dir - Camera.CFrame.RightVector end
    if UIS:IsKeyDown(Enum.KeyCode.D) then dir = dir + Camera.CFrame.RightVector end
    if UIS:IsKeyDown(Enum.KeyCode.Space) then dir = dir + Vector3.new(0, 1, 0) end
    if UIS:IsKeyDown(Enum.KeyCode.LeftShift) then dir = dir - Vector3.new(0, 1, 0) end
    State.flyVelocity.Velocity = dir * State.flySpeed
end)

-- Anti Ragdoll
RunService.Heartbeat:Connect(function()
    if not State.antiRagdoll then return end
    local char = LP.Character
    if not char then return end
    local hum = char:FindFirstChildOfClass("Humanoid")
    local root = char:FindFirstChild("HumanoidRootPart")
    if not hum or not root then return end
    local st = hum:GetState()
    if st == Enum.HumanoidStateType.Physics or st == Enum.HumanoidStateType.Ragdoll or st == Enum.HumanoidStateType.FallingDown then
        pcall(function()
            hum:ChangeState(Enum.HumanoidStateType.GettingUp)
            hum.PlatformStand = false
            hum.Sit = false
            root.AssemblyLinearVelocity = Vector3.zero
            root.AssemblyAngularVelocity = Vector3.zero
            workspace.CurrentCamera.CameraSubject = hum
        end)
    end
end)

local function makeToggle(parent, label, default, callback)
    local row = Instance.new("Frame")
    row.Size = UDim2.new(1, 0, 0, 38)
    row.BackgroundColor3 = Color3.fromRGB(22, 22, 22)
    row.BorderSizePixel = 0
    row.Parent = parent
    local rc = Instance.new("UICorner")
    rc.CornerRadius = UDim.new(0, 8)
    rc.Parent = row
    local rs = Instance.new("UIStroke")
    rs.Color = Color3.fromRGB(60, 60, 60)
    rs.Thickness = 1
    rs.Parent = row
    local lbl = Instance.new("TextLabel")
    lbl.Size = UDim2.new(0.65, 0, 1, 0)
    lbl.Position = UDim2.new(0, 12, 0, 0)
    lbl.BackgroundTransparency = 1
    lbl.Text = label
    lbl.TextColor3 = Color3.fromRGB(255, 255, 255)
    lbl.Font = Enum.Font.GothamBold
    lbl.TextSize = 11
    lbl.TextXAlignment = Enum.TextXAlignment.Left
    lbl.Parent = row
    local track = Instance.new("Frame")
    track.Size = UDim2.new(0, 34, 0, 18)
    track.Position = UDim2.new(1, -46, 0.5, -9)
    track.BackgroundColor3 = default and Color3.fromRGB(255, 255, 255) or Color3.fromRGB(45, 45, 45)
    track.BorderSizePixel = 0
    track.Parent = row
    local tc = Instance.new("UICorner")
    tc.CornerRadius = UDim.new(0, 9)
    tc.Parent = track
    local knob = Instance.new("Frame")
    knob.Size = UDim2.new(0, 14, 0, 14)
    knob.Position = default and UDim2.new(1, -16, 0.5, -7) or UDim2.new(0, 2, 0.5, -7)
    knob.BackgroundColor3 = default and Color3.fromRGB(0, 0, 0) or Color3.fromRGB(255, 255, 255)
    knob.BorderSizePixel = 0
    knob.Parent = track
    local kc = Instance.new("UICorner")
    kc.CornerRadius = UDim.new(0, 7)
    kc.Parent = knob
    local state = default
    local click = Instance.new("TextButton")
    click.Size = UDim2.new(1, 0, 1, 0)
    click.BackgroundTransparency = 1
    click.Text = ""
    click.Parent = row
    click.MouseButton1Click:Connect(function()
        state = not state
        track.BackgroundColor3 = state and Color3.fromRGB(255, 255, 255) or Color3.fromRGB(45, 45, 45)
        knob.Position = state and UDim2.new(1, -16, 0.5, -7) or UDim2.new(0, 2, 0.5, -7)
        knob.BackgroundColor3 = state and Color3.fromRGB(0, 0, 0) or Color3.fromRGB(255, 255, 255)
        rs.Color = state and Color3.fromRGB(255, 255, 255) or Color3.fromRGB(60, 60, 60)
        if callback then callback(state) end
    end)
end

    lbl.TextXAlignment = Enum.TextXAlignment.Left

    box.Size = UDim2.new(0, 60, 0, 24)
    box.Position = UDim2.new(1, -72, 0.5, -12)
    box.BackgroundColor3 = Color3.fromRGB(14, 14, 14)

    box.Parent = row

-- Drag
local dragging, dragStart, startPos
Header.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        dragging = true
        dragStart = input.Position
        startPos = Main.Position
    end
end)
UIS.InputChanged:Connect(function(input)
    if not dragging then return end
    if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
        local delta = input.Position - dragStart
        Main.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
    end
end)
UIS.InputEnded:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        dragging = false
    end
end)
